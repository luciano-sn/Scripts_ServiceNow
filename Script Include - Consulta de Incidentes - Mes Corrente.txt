var Consulta_Incidente_Mes = Class.create();
Consulta_Incidente_Mes.prototype = {
    initialize: function() {
    },
	consultaInc: function () {
        var incGR = new GlideRecord('incident');
        incGR.addEncodedQuery('sys_created_onONThis month@javascript:gs.beginningOfThisMonth()@javascript:gs.endOfThisMonth()');
        incGR.query();

        while (incGR.next()) {
            gs.info('Chamado: ' + incGR.getValue('number'));
            gs.info('Aberto por: ' + incGR.getDisplayValue('caller_id'));
            gs.info('Status: ' + incGR.getDisplayValue('state'));
            gs.info('Prioridade: ' + incGR.getDisplayValue('urgency'));
        }
    },

    type: 'Consulta_Incidente_Mes'
};
